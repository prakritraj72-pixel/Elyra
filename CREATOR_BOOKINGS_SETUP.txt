ELYRA — Creator booking management security

Run this SQL once in the Supabase SQL Editor:

alter table public.bookings enable row level security;

drop policy if exists "Creators can view own booking requests" on public.bookings;
create policy "Creators can view own booking requests"
on public.bookings
for select
to authenticated
using (creator_id = auth.uid());

-- Allow creators to update their own booking rows, but the trigger below limits
-- creator updates to status + updated_at only.
drop policy if exists "Creators can update own pending booking status" on public.bookings;
create policy "Creators can update own pending booking status"
on public.bookings
for update
to authenticated
using (
  creator_id = auth.uid()
  and status = 'pending'
)
with check (
  creator_id = auth.uid()
  and status in ('accepted', 'rejected')
);

-- Prevent a creator from changing amount, date, customer, address, notes, etc.
-- while using the booking-status update policy.
create or replace function public.protect_creator_booking_update()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if auth.uid() = old.creator_id then
    if old.status <> 'pending' then
      raise exception 'Only pending bookings can be changed';
    end if;

    if new.status not in ('accepted', 'rejected') then
      raise exception 'Invalid booking status';
    end if;

    if (to_jsonb(new) - 'status' - 'updated_at')
       <> (to_jsonb(old) - 'status' - 'updated_at') then
      raise exception 'Creators may only change booking status';
    end if;
  end if;

  return new;
end;
$$;

drop trigger if exists protect_creator_booking_update on public.bookings;
create trigger protect_creator_booking_update
before update on public.bookings
for each row
execute function public.protect_creator_booking_update();

Important: the creator-bookings.html page now uses the normal Supabase update call. The RLS policy + trigger make that update safe: creators can only move their own pending booking from pending → accepted/rejected and cannot alter price, date, address, customer data, or other fields.
