ELYRA — Creator booking management security

Run this SQL once in the Supabase SQL Editor:

alter table public.bookings enable row level security;

drop policy if exists "Creators can view own booking requests" on public.bookings;
create policy "Creators can view own booking requests"
on public.bookings
for select
to authenticated
using (creator_id = auth.uid());

create or replace function public.creator_set_booking_status(
  p_booking_id uuid,
  p_status text
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_booking public.bookings%rowtype;
begin
  if auth.uid() is null then
    raise exception 'Not authenticated';
  end if;

  if p_status not in ('accepted', 'rejected') then
    raise exception 'Invalid booking status';
  end if;

  select * into v_booking
  from public.bookings
  where id = p_booking_id
    and creator_id = auth.uid()
  for update;

  if not found then
    raise exception 'Booking not found';
  end if;

  if v_booking.status <> 'pending' then
    raise exception 'Only pending bookings can be changed';
  end if;

  update public.bookings
  set status = p_status, updated_at = now()
  where id = p_booking_id
    and creator_id = auth.uid()
    and status = 'pending';

  return jsonb_build_object('id', p_booking_id, 'status', p_status);
end;
$$;

revoke all on function public.creator_set_booking_status(uuid, text) from public;
grant execute on function public.creator_set_booking_status(uuid, text) to authenticated;

Important: the creator booking page should call this RPC for Accept/Reject. Do not add a broad creator UPDATE policy, because that would let a creator modify amount, date, address, or other booking fields.
